MTS.SortableList

Drag-and-drop sortable list — numbered, move buttons, icons, avatars, badges, read-only.

1 — Workflow de aprobaciones — numbered + moveButtons + disabled
— reordena la lista —
<div id="my-list"></div> <script> new MTS.SortableList('#my-list', { numbered: true, moveButtons: true, items: [ { id:'1', title:'Design review', badge:{ label:'Pending', variant:'warning' } }, { id:'2', title:'Backend API', badge:{ label:'In progress', variant:'primary' } }, { id:'3', title:'QA testing', disabled: true }, ], // Fires when reordered / Se dispara al reordenar onReorder: (e) => { console.log(e.detail.items); // → new order console.log(e.detail.fromIndex); // → 0 console.log(e.detail.toIndex); // → 2 }, // Fires on item click / Se dispara al hacer click onItemClick: (e) => console.log(e.detail.item.id), }); </script>
new MTS.SortableList('#my-list', { variant: 'default', numbered: true, // show order numbers / mostrar números de orden moveButtons: true, // show ↑ ↓ buttons / mostrar botones ↑ ↓ showHandle: true, // show drag handle / mostrar asa de arrastre items: [ { id:'1', title:'Design review', description:'Ana García', badge:{ label:'Pending', variant:'warning' } }, { id:'2', title:'Backend API', description:'Pedro López', badge:{ label:'In progress', variant:'primary' } }, { id:'3', title:'QA testing', description:'Laura Sánchez', disabled: true }, { id:'4', title:'Deploy to prod', description:'Carlos Ruiz', badge:{ label:'Blocked', variant:'danger' } }, { id:'5', title:'Write changelog', description:'María Torres', badge:{ label:'Todo', variant:'default' } }, ], onReorder: (e) => console.log('reordered from', e.detail.fromIndex, 'to', e.detail.toIndex), onItemClick: (e) => console.log('clicked:', e.detail.item.title), });

4 — API — addItem · removeItem · lock · unlock
— usa los botones —
<div id="my-list"></div> <script> const list = new MTS.SortableList('#my-list', { items:[...] }); list.addItem({ id:'new', title:'New item' }) list.addItem({ id:'top', title:'At top' }, 0) list.removeItem('new') list.updateItem('top', { title:'Updated' }) list.lock() // disable drag / deshabilitar drag list.unlock() // re-enable / rehabilitar list.getItems() // → current order </script>
const list = new MTS.SortableList('#my-list', { numbered: true, items: [ { id:'sl-a1', title:'Item Alpha' }, { id:'sl-a2', title:'Item Beta' }, { id:'sl-a3', title:'Item Gamma' }, ], onReorder: (e) => console.log('reordered:', e.detail.items.map(i => i.id)), }); list.addItem({ id:'new-item', title:'New item' }) // add at end list.addItem({ id:'top-item', title:'At top' }, 0) // add at index list.removeItem('sl-a1') // remove by id list.updateItem('sl-a2', { title:'Updated title' }) // update item list.lock() // read-only list.unlock() // re-enable drag list.getItems() // current order
