2 — Selectable + onSelect
— haz click en un nodo —
<div id="my-tree"></div> <script> new MTS.Tree('#my-tree', { selectable: true, nodes: [...], // Fires when a node is selected / Se dispara al seleccionar un nodo onSelect: (e) => { console.log(e.detail.node.id); // → 'src' console.log(e.detail.path); // → ['root', 'src'] }, }); </script>
new MTS.Tree('#my-tree', { selectable: true, // click to select / click para seleccionar showLines: true, nodes: [ { id:'dashboard', label:'Dashboard', selected: true }, { id:'analytics', label:'Analytics' }, { id:'reports', label:'Reports', children:[ { id:'monthly', label:'Monthly report' }, { id:'annual', label:'Annual report' }, ]}, { id:'settings', label:'Settings', children:[ { id:'profile', label:'Profile' }, { id:'security', label:'Security' }, { id:'billing', label:'Billing', disabled: true }, ]}, ], // Fires when a node is selected / Se dispara al seleccionar un nodo onSelect: (e) => { console.log(e.detail.node.id); // → 'dashboard' console.log(e.detail.path); // → ['dashboard'] }, });
3 — Checkable con propagación a hijos / Checkboxes with child propagation
— marca/desmarca checkboxes —
<div id="my-tree"></div> <script> new MTS.Tree('#my-tree', { checkable: true, nodes: [...], // Fires when checkbox changes / Se dispara al cambiar un checkbox onCheck: (e) => { console.log(e.detail.node.id); // → 'users-read' console.log(e.detail.checked); // → true | false console.log(e.detail.checkedIds); // → ['users-read', 'reports-view'] }, }); </script>
new MTS.Tree('#my-tree', { checkable: true, // show checkboxes / mostrar checkboxes expandAll: true, // expand all on init / expandir todo al inicio showIcons: false, nodes: [ { id:'users', label:'Users', children:[ { id:'users-read', label:'Read', checked:true }, { id:'users-write', label:'Write', checked:true }, { id:'users-delete', label:'Delete', disabled:true }, ]}, { id:'reports', label:'Reports', children:[ { id:'reports-view', label:'View' }, { id:'reports-export', label:'Export' }, ]}, { id:'settings', label:'Settings', children:[ { id:'settings-read', label:'Read' }, { id:'settings-write', label:'Write', checked:true }, ]}, ], // Fires when checkbox changes / Se dispara al cambiar un checkbox onCheck: (e) => { console.log('checked IDs:', e.detail.checkedIds); }, });
4 — Badges + íconos custom + API
<div id="my-tree"></div> <script> const tree = new MTS.Tree('#my-tree', { selectable: true, nodes: [ { id:'eng', label:'Engineering', badge:{ label:'12', variant:'primary' }, children:[ { id:'frontend', label:'Frontend', badge:{ label:'4' } }, { id:'backend', label:'Backend', badge:{ label:'8' } }, ]}, ], }); tree.expandAll() tree.collapseAll() tree.select('frontend') tree.deselect() tree.getChecked() // → checked IDs </script>
const tree = new MTS.Tree('#my-tree', { selectable: true, showLines: true, nodes: [ { id:'eng', label:'Engineering', badge:{ label:'12', variant:'primary' }, children:[ { id:'frontend', label:'Frontend', badge:{ label:'4', variant:'info' } }, { id:'backend', label:'Backend', badge:{ label:'8', variant:'success' } }, ]}, { id:'design', label:'Design', badge:{ label:'3', variant:'warning' } }, { id:'product', label:'Product', badge:{ label:'5', variant:'default' } }, ], onSelect: (e) => console.log('selected:', e.detail.node.id), }); // API tree.expandAll() // expand all / expandir todos tree.collapseAll() // collapse all / colapsar todos tree.expand('eng') // expand one / expandir uno tree.collapse('eng') // collapse one / colapsar uno tree.select('frontend') // select programmatically / seleccionar programáticamente tree.deselect() // deselect all / deseleccionar tree.getChecked() // → array of checked IDs
