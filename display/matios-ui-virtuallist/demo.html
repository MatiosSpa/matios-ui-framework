2 — appendItems() — Infinite scroll simulado / Simulated infinite scroll
— haz scroll hasta el final para cargar más —
<div id="my-list"></div> <script> let page = 1; const list = new MTS.VirtualList('#my-list', { items: fetchPage(page++), renderItem: (item) => `<div>${item.title}</div>`, // Fires when near the end / Se dispara cerca del final del scroll onEndReached: async (e) => { console.log('total so far:', e.detail.total); const more = await fetchPage(page++); list.appendItems(more); }, }); </script>
function fetchPage(page) { // Simulate API call / Simular llamada API return Array.from({ length: 20 }, (_, i) => ({ id: page * 20 + i, title: `Post #${page * 20 + i + 1}`, date: `Dec ${(i % 28) + 1}, 2024`, })); } let page = 1; const list = new MTS.VirtualList('#my-list', { items: fetchPage(page++), itemHeight: 56, height: 400, endThreshold: 150, // px from bottom to trigger / px desde el fondo para activar renderItem: (item, index) => { const el = document.createElement('div'); el.className = 'vl-row'; el.innerHTML = ` <span class="vl-idx">#${index + 1}</span> <span class="vl-name">${item.title}</span> <span class="vl-meta">${item.date}</span>`; return el; }, // Fires when near the end — load more / Se dispara cerca del final — cargar más onEndReached: (e) => { console.log('loading page', page, '(', e.detail.total, 'items so far)'); const more = fetchPage(page++); list.appendItems(more); }, });
