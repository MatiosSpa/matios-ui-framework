const sp = new MTS.Splitter('#my-splitter', { direction: 'horizontal', initialSize: 35, // first panel starts at 35% minSize: 20, // minimum 20% / mínimo 20% maxSize: 80, // maximum 80% / máximo 80% // Fires while dragging / Se dispara mientras se arrastra onChange: (e) => { console.log(e.detail.firstSize); // → 42.5 console.log(e.detail.secondSize); // → 57.5 }, // Fires when drag starts / Se dispara al iniciar el arrastre onDragStart: (e) => console.log('drag started'), // Fires when drag ends / Se dispara al terminar el arrastre onDragEnd: (e) => { console.log('final size:', e.detail.firstSize + '%'); saveLayout(e.detail); }, }); // API sp.setSize(40) // set first panel size / establecer tamaño del primer panel sp.getSizes() // → { firstSize: 40, secondSize: 60 } sp.collapse() // collapse first panel / colapsar primer panel sp.expand() // expand / expandir

const sp = new MTS.Splitter('#my-splitter', { direction: 'vertical', initialSize: 55, collapsible: true, // double click handle to collapse / doble click en divisor para colapsar gutterSize: '8px', minSize: 15, maxSize: 85, onChange: (e) => console.log(e.detail.firstSize + '%'), onDragStart: (e) => console.log('drag started'), onDragEnd: (e) => { console.log('saved:', e.detail.firstSize + '%'); localStorage.setItem('split-size', e.detail.firstSize); }, }); // Restore saved size / Restaurar tamaño guardado const saved = localStorage.getItem('split-size'); if (saved) sp.setSize(+saved); // API sp.collapse() // collapse top panel sp.expand() // restore sp.getSizes() // → { firstSize: 55, secondSize: 45 }
