— arrastra tarjetas o haz click en una —
<div id="my-board"></div> <script> const board = new MTS.Kanban('#my-board', { addCards: true, columns: [ { id:'backlog', title:'Backlog', cards:[ { id:'c1', title:'Set up CI/CD', priority:'high', tags:['devops'], assignees:[{ name:'Ana García' }], dueDate:'Dec 15' }, ]}, { id:'in-progress', title:'In Progress', wip:3, cards:[] }, { id:'done', title:'Done', color:'#16a34a', cards:[] }, ], // Fires on card move / Se dispara al mover una tarjeta onMove: (e) => { console.log(e.detail.card.id); console.log(e.detail.fromColId, '→', e.detail.toColId); }, // Fires on card click / Se dispara al hacer click en una tarjeta onCardClick: (e) => openDetail(e.detail.card), // Fires on card add / Se dispara al agregar tarjeta onAddCard: (e) => console.log(e.detail.card, e.detail.colId), }); </script>

2 — API — addCard · moveCard · removeCard
— usa los botones —
<script> const board = new MTS.Kanban('#my-board', { columns:[...] }); board.addCard('backlog', { id:'new-1', title:'New task', priority:'medium' }); board.moveCard('new-1', 'done'); board.removeCard('new-1'); board.getCards(); // all cards / todas las tarjetas board.getCards('backlog'); // cards in column / tarjetas de la columna </script>
