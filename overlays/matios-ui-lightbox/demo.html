MTS.Lightbox

Media viewer — images, video, YouTube, Vimeo · zoom · thumbnails · auto-bind.

1 — Uso programático / Programmatic
— abre el lightbox —
<button onclick="lb.open(0)">Open gallery</button> <script> const lb = new MTS.Lightbox([ { src: '/img/photo1.jpg', caption: 'Mountain view' }, { src: '/img/photo2.jpg', caption: 'Ocean sunset' }, { src: '/img/photo3.jpg', caption: 'Forest trail' }, ], { loop: true, zoom: true, thumbnails: true, download: true, // Fires when lightbox opens / Se dispara al abrir onOpen: (e) => console.log('opened:', e.detail.index), // Fires when item changes / Se dispara al cambiar el ítem onChange: (e) => console.log('changed:', e.detail.index), // Fires when closed / Se dispara al cerrar onClose: () => console.log('closed'), }); lb.open(0); </script>
const lb = new MTS.Lightbox([ { src: 'https://picsum.photos/seed/1/1200/800', caption: 'Landscape 1', thumb: 'https://picsum.photos/seed/1/120/80', type: 'image', }, { src: 'https://picsum.photos/seed/2/1200/800', caption: 'Landscape 2', thumb: 'https://picsum.photos/seed/2/120/80', }, { src: 'https://picsum.photos/seed/3/1200/800', caption: 'Landscape 3', thumb: 'https://picsum.photos/seed/3/120/80', }, ], { loop: true, zoom: true, thumbnails: true, download: false, animation: 'fade', // 'fade' | 'slide' // Fires when lightbox opens / Se dispara al abrir el lightbox onOpen: (e) => { console.log(e.detail.item); // → { src, caption, ... } console.log(e.detail.index); // → 0 }, // Fires when item changes / Se dispara al cambiar el ítem activo onChange: (e) => console.log('index:', e.detail.index), // Fires when closed / Se dispara al cerrar onClose: () => console.log('closed'), }); lb.open(0); // open at first item / abrir en el primer ítem lb.open(2); // open at third item / abrir en el tercer ítem
2 — Auto-bind con selector CSS / Auto-bind from CSS selector
🏔️ 🌊 🌲 🏜️ 🌅 🌺
<!-- Add data-lightbox to any element / Agregar data-lightbox a cualquier elemento --> <a href="/img/1.jpg" data-lightbox data-caption="Mountain view"> <img src="/img/1-thumb.jpg" alt="Mountain"> </a> <a href="/img/2.jpg" data-lightbox data-caption="Ocean sunset"> <img src="/img/2-thumb.jpg" alt="Ocean"> </a> <script> // Pass selector — clicks bound automatically / Selector — clicks vinculados automáticamente new MTS.Lightbox('[data-lightbox]', { thumbnails: true, zoom: true, onChange: (e) => console.log(e.detail.index), }); </script>
// CSS selector mode — reads data-* attributes from matched elements // Modo selector CSS — lee atributos data-* de los elementos encontrados new MTS.Lightbox('[data-lightbox]', { thumbnails: true, zoom: true, loop: true, // Fires when item changes / Se dispara al cambiar el ítem activo onChange: (e) => console.log('active:', e.detail.index), }); // data-* attributes read automatically / Atributos data-* leídos automáticamente: // data-src → item.src (or href attribute) // data-type → item.type ('image' | 'video' | 'youtube' | 'vimeo') // data-caption → item.caption // data-alt → item.alt // data-thumb → item.thumb
3 — API + opciones / Options + API
— usa los botones —
<script> const lb = new MTS.Lightbox([...], { animation: 'slide', // 'fade' | 'slide' download: true, // show download button / mostrar botón descarga counter: true, // show "1 / 3" counter / mostrar contador }); lb.open(0) // open at index / abrir en índice lb.close() // close / cerrar lb.next() // next item / ítem siguiente lb.prev() // prev item / ítem anterior lb.goTo(2) // jump to index / saltar al índice lb.on('change', (e) => console.log(e.detail.index)) lb.off('change', handler) lb.destroy() </script>
const lb = new MTS.Lightbox([ { src: 'https://picsum.photos/seed/100/1200/800', caption: 'Image 1' }, { src: 'https://picsum.photos/seed/200/1200/800', caption: 'Image 2' }, { src: 'https://picsum.photos/seed/300/1200/800', caption: 'Image 3' }, ], { animation: 'slide', // 'fade' | 'slide' download: true, // show download button / botón de descarga counter: true, // show "1 / 3" / mostrar contador thumbnails: false, loop: true, onOpen: (e) => console.log('opened:', e.detail.index), onChange: (e) => console.log('changed:', e.detail.index), onClose: () => console.log('closed'), }); // API lb.open(0) // open at index / abrir en índice lb.close() // close / cerrar lb.next() // next / siguiente lb.prev() // previous / anterior lb.goTo(2) // jump / saltar // Events / Eventos lb.on('change', (e) => console.log(e.detail.index)) lb.off('change', handler) lb.destroy()
