MTS.Alert

Alertas inline con variantes, íconos, acciones y auto-dismiss.

1 — Variantes
MTS.Alert.show('#el', {
  variant: 'info',      // 'info'|'success'|'warning'|'danger'
  title:   'Información',
  message: 'El sistema se actualizará el próximo domingo.',
  closable: true,
  onClose: (e) => console.log('cerrado'),
})
// También con acción
MTS.Alert.show('#el', {
  variant: 'success',
  title:   'Guardado',
  message: 'Los cambios ya están disponibles.',
  action:  'Ver cambios',
  onAction: () => console.log('acción ejecutada'),
})
2 — Banner (ancho completo)
MTS.Alert.show('#el', {
  variant: 'info',
  banner:   true,     // sin border-radius, diseño más compacto
  message: 'Versión 2.0 disponible. Nuevos componentes y mejoras.',
  action:  'Ver novedades',
  onAction: () => console.log('ver novedades'),
})
3 — Programático con auto-dismiss
— dispara una alerta —
// Mostrar en un contenedor (reemplaza el anterior)
MTS.Alert.show('#mi-contenedor', {
  variant:     'success',
  title:       'Guardado',
  message:     'Los datos se guardaron correctamente.',
  autoDismiss: 3000,    // se cierra automáticamente en 3 segundos
  onClose: (e) => console.log('cerrado:', e.detail),
})
4 — Sin ícono / sin botón cerrar
MTS.Alert.show('#el', {
  variant: 'warning',
  icon:     false,    // sin ícono
  message: 'Sin ícono y sin botón de cierre.',
  closable: false,
})
HTML declarativo
<div id="miAlert" data-variant="success" data-title="¡Guardado!" data-message="Cambios guardados." data-closable></div>
<script>new MTS.Alert('#miAlert', { onClose: () => console.log('cerrado') })</script>
