3 — Via JS — contador, removible, dot
— usa los botones o remueve un tag —
<span id="badge-count"></span> <span id="tag-ts"></span> <span id="badge-dot"></span> <script> // Dynamic counter / Contador dinámico const badge = new MTS.Badge('#badge-count', { count: 42, variant: 'danger', }); badge.setCount(100); // → shows "99+" badge.setCount(0); // → hides badge // Removable tag / Tag removible new MTS.Badge('#tag-ts', { label: 'TypeScript', variant: 'primary', removable: true, // Fires on remove / Se dispara al remover onRemove: (e) => e.detail.badge._el.remove(), }); // Notification dot / Dot de notificación new MTS.Badge('#badge-dot', { dot: true, pulse: true, variant: 'danger', }); </script>
// Dynamic counter / Contador dinámico const badge = new MTS.Badge('#badge-count', { count: 42, maxCount: 99, // shows "99+" above 99 / muestra "99+" si supera variant: 'danger', shape: 'pill', size: 'md', }); badge.setCount(100); // → "99+" badge.setCount(0); // → hides / oculta badge.setCount(5); // → "5" // Removable tags / Tags removibles ['TypeScript','React','CSS'].forEach(label => { new MTS.Badge('#tag-' + label.toLowerCase(), { label: label, variant: 'primary', removable: true, // Fires when remove is clicked / Se dispara al hacer click en remover onRemove: (e) => { e.detail.badge._el.remove(); console.log('removed:', label); }, }); }); // Notification dot with pulse / Dot de notificación con pulso new MTS.Badge('#badge-dot', { dot: true, pulse: true, variant: 'danger', });
