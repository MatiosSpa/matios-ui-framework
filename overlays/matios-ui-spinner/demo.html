1 — Variantes
circle
ring
dots
bars
pulse
new MTS.Spinner('#el', { variant:'circle' })
new MTS.Spinner('#el', { variant:'ring'   })
new MTS.Spinner('#el', { variant:'dots'   })
new MTS.Spinner('#el', { variant:'bars'   })
new MTS.Spinner('#el', { variant:'pulse'  })
2 — Tamaños
xs
sm
md
lg
xl
3 — Con label y colores custom
new MTS.Spinner('#el', { variant:'circle', label:'Cargando...' })
new MTS.Spinner('#el', { variant:'dots',   label:'Procesando datos', size:'lg' })
new MTS.Spinner('#el', { variant:'ring',   color:'#f59e0b' })
new MTS.Spinner('#el', { variant:'bars',   color:'#34d399' })

<script>new MTS.Spinner('#miSpinner')</script>
