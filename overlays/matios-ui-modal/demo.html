MTS.Modal

Modales con tamaños, botones configurables y helpers rápidos.

1 — Básico y tamaños
— abre un modal —
const modal = new MTS.Modal({
  title: 'Título del modal',
  body:  '<p>Contenido HTML...</p>',
  size:  'md',    // 'sm'|'md'|'lg'|'xl'|'fullscreen'
  buttons: [
    { label:'Cancelar', variant:'ghost',   close:true },
    { label:'Aceptar',  variant:'primary', close:true,
      onClick: () => console.log('aceptar') },
  ],
  onShow:   () => console.log('shown'),
  onHide:   () => console.log('hidden'),
})
modal.show()
modal.hide()
2 — Helpers rápidos
— usa los helpers rápidos —
// Confirmación rápida
MTS.Modal.confirm({
  title:     '¿Eliminar registro?',
  message:   'Esta acción no se puede deshacer.',
  variant:   'danger',
  onConfirm: () => console.log('confirmado'),
  onCancel:  () => console.log('cancelado'),
})

// Alerta informativa
MTS.Modal.alert({
  title:   'Guardado',
  message: 'Los cambios se guardaron correctamente.',
  variant: 'success',
})

// Prompt — input de texto
MTS.Modal.prompt({
  title:       'Renombrar archivo',
  placeholder: 'Nuevo nombre...',
  onConfirm: (value) => console.log('valor:', value),
})
3 — Con formulario
— abre el modal con formulario —
const modal = new MTS.Modal({
  title: 'Crear carpeta',
  body: `<div style="display:flex;flex-direction:column;gap:12px">
    <label>Nombre</label>
    <input id="modal-input" type="text" placeholder="Mi carpeta">
  </div>`,
  buttons: [
    { label:'Cancelar', variant:'ghost',   close:true },
    { label:'Crear',    variant:'primary', close:true,
      onClick: () => {
        const val = document.getElementById('modal-input').value
        console.log('nombre:', val)
      }},
  ],
  onShow: () => setTimeout(() => document.getElementById('modal-input')?.focus(), 100),
})
4 — Static (no cierra con Esc/backdrop)
// Static: no cierra con Esc ni click en backdrop
new MTS.Modal({
  title:  'Aceptar términos',
  static:  true,
  body:   '...',
  buttons: [{ label:'Aceptar', variant:'primary', close:true }],
})

// Scrollable: body con scroll interno
new MTS.Modal({
  title:      'Lista larga',
  scrollable:  true,
  body:       '...',  // contenido largo
})
