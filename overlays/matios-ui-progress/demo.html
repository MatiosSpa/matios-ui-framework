MTS.Progress

Bar · circle · indeterminate — variants, striped, animated, smooth value transitions.

1 — Barras lineales con variantes / Linear bars with variants
<div id="my-progress"></div> <script> new MTS.Progress('#my-progress', { type: 'bar', value: 65, variant: 'primary', // Fires on value change / Se dispara al cambiar el valor onChange: (e) => { console.log(e.detail.value); // → 65 console.log(e.detail.pct); // → 0.65 }, // Fires when value reaches max / Se dispara al llegar al máximo onComplete: (e) => console.log('Done!'), }); </script>
// All variants / Todas las variantes ['default','primary','success','warning','danger','info'].forEach((v, i) => { new MTS.Progress('#bar-' + v, { type: 'bar', value: (i + 1) * 15, variant: v, showValue: true, }); }); // With label / Con label new MTS.Progress('#bar-label', { type: 'bar', value: 3, max: 10, showLabel: true, labelFormat: (value, pct) => `Step ${value} of 10`, variant: 'primary', });

// Indeterminate loading / Carga indeterminada new MTS.Progress('#loading', { type: 'indeterminate', variant: 'primary', }); // API usage / Uso de la API const prog = new MTS.Progress('#my-bar', { type: 'bar', value: 0, // Fires on value change / Se dispara al cambiar el valor onChange: (e) => { console.log(e.detail.value, e.detail.pct); }, // Fires when reaches 100 / Se dispara al llegar al 100 onComplete: (e) => console.log('Done!'), }); prog.setValue(75) // animate to 75 / animar hasta 75 prog.setValue(75, false) // no animation / sin animación prog.increment(10) // +10 prog.decrement(5) // -5 prog.setVariant('success') // change color / cambiar color prog.reset() // back to min / volver al mínimo
