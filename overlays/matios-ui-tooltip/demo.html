MTS.Tooltip

Smart positioning tooltip — hover, click, focus triggers · dark/light variants · HTML content.

1 — Posiciones / Positions
<button id="btn-save">Save</button> <script> new MTS.Tooltip('#btn-save', { content: 'Save document', position: 'top', // 'top' | 'bottom' | 'left' | 'right' trigger: 'hover', // default }); </script>
new MTS.Tooltip('#btn-top', { content: 'Tooltip on top', position: 'top' }); new MTS.Tooltip('#btn-bottom', { content: 'Tooltip on bottom', position: 'bottom' }); new MTS.Tooltip('#btn-left', { content: 'Tooltip on left', position: 'left' }); new MTS.Tooltip('#btn-right', { content: 'Tooltip on right', position: 'right' });
2 — Triggers
<button id="btn-hover">Hover</button> <button id="btn-click">Click</button> <input id="input-focus" placeholder="Focus me"> <script> new MTS.Tooltip('#btn-hover', { content: 'Hover tooltip', trigger: 'hover' }); new MTS.Tooltip('#btn-click', { content: 'Click tooltip', trigger: 'click' }); new MTS.Tooltip('#input-focus', { content: 'Focus tooltip', trigger: 'focus' }); </script>
// Hover (default) / Hover (por defecto) new MTS.Tooltip('#btn-hover', { content: 'Hover to see me', trigger: 'hover', }); // Click — toggles / Click — alterna new MTS.Tooltip('#btn-click', { content: 'Click again to hide', trigger: 'click', }); // Focus — shows on focus / Se muestra al enfocar new MTS.Tooltip('#input-focus', { content: 'Enter your email address', trigger: 'focus', position: 'bottom', });
3 — Variantes y contenido HTML / Variants and HTML content
<!-- Dark (default) / Oscuro --> <button data-tooltip="Dark tooltip">Dark</button> <!-- Light / Claro --> <button data-tooltip="Light tooltip" data-tooltip-variant="light">Light</button> <!-- HTML content / Contenido HTML --> <button id="btn-html">Con HTML</button> <script> new MTS.Tooltip('#btn-html', { content: '<strong>Pro tip:</strong> Use ⌘S to save.', maxWidth: 200, }); </script>
// Dark variant (default) / Variante oscura new MTS.Tooltip('#btn-dark', { content: 'Dark tooltip', variant: 'dark' }); // Light variant / Variante clara new MTS.Tooltip('#btn-light', { content: 'Light tooltip', variant: 'light' }); // HTML content / Contenido HTML new MTS.Tooltip('#btn-html', { content: '<strong>Pro tip:</strong> Use ⌘S to save quickly.', maxWidth: 200, position: 'bottom', }); // Custom colors / Colores personalizados new MTS.Tooltip('#btn-custom', { content: 'Custom violet tooltip', bg: '#7c3aed', color: '#ffffff', }); // With delay / Con delay new MTS.Tooltip('#btn-delay', { content: 'Appears after 500ms', delay: 500, hideDelay: 200, });
4 — initAll() — desde atributos HTML
<!-- Add data-tooltip to any element / Agregar data-tooltip a cualquier elemento --> <button data-tooltip="Save document" data-tooltip-position="top">Save</button> <button data-tooltip="Delete item" data-tooltip-position="bottom" data-tooltip-variant="light">Delete</button> <button data-tooltip="Click to toggle" data-tooltip-trigger="click">Info</button> <script> // Initialize all at once / Inicializar todos a la vez MTS.Tooltip.initAll(); </script>
// Initialize all [data-tooltip] elements at once // Inicializa todos los elementos [data-tooltip] a la vez MTS.Tooltip.initAll(); // Or within a specific container / O dentro de un contenedor específico MTS.Tooltip.initAll('#my-toolbar'); // Supported data-* attributes / Atributos data-* soportados: // data-tooltip → content // data-tooltip-position → 'top' | 'bottom' | 'left' | 'right' // data-tooltip-trigger → 'hover' | 'click' | 'focus' // data-tooltip-variant → 'dark' | 'light' // data-tooltip-delay → ms // data-tooltip-max-width → px
