MTS.Infinite

Infinite scroll with IntersectionObserver — vertical, grid and table layouts. Dev controls data loading.

1 — Layout vertical (default)
— haz scroll hasta el final para cargar más —
<div id="my-list"></div> <script> new MTS.Infinite('#my-list', { pageSize: 15, layout: 'vertical', // Required: async data loader / Requerido: cargador de datos async onLoadMore: async ({ page, pageSize }) => { const res = await fetch(`/api/users?page=${page}&size=${pageSize}`); const data = await res.json(); return { items: data.users, hasMore: data.hasNext }; }, // Required: item renderer / Requerido: renderizador de ítems renderItem: (user, index) => ` <div class="row"> <span>#${index + 1}</span> <span>${user.name}</span> </div>`, // Fires after each load / Se dispara tras cada carga onLoad: (e) => console.log('loaded page', e.detail.page), // Fires when all data is loaded / Se dispara al no haber más datos onEnd: (e) => console.log('done, total:', e.detail.total), }); </script>
new MTS.Infinite('#my-list', { pageSize: 15, layout: 'vertical', // Data loader — receives { page, pageSize }, must return { items, hasMore } // Cargador de datos — recibe { page, pageSize }, retorna { items, hasMore } onLoadMore: async ({ page, pageSize }) => { // Simulate API / Simular API await new Promise(r => setTimeout(r, 300)); const items = Array.from({ length: pageSize }, (_, i) => ({ id: (page - 1) * pageSize + i, name: `Usuario ${(page - 1) * pageSize + i + 1}`, email: `user${(page - 1) * pageSize + i}@empresa.com`, })); return { items, hasMore: page < 5 }; // 5 pages max }, // Item renderer / Renderizador de ítems renderItem: (user, index) => { const el = document.createElement('div'); el.className = 'inf-row'; el.innerHTML = `<span class="inf-idx">#${index + 1}</span> <span class="inf-name">${user.name}</span> <span class="inf-email">${user.email}</span>`; return el; }, // Fires after each load / Se dispara tras cada carga onLoad: (e) => console.log(`Page ${e.detail.page}: ${e.detail.items.length} items`), // Fires when no more data / Se dispara al no haber más datos onEnd: (e) => console.log(`Done — ${e.detail.total} total`), });

const grid = new MTS.Infinite('#my-grid', { layout: 'grid', // 3-column grid / grid de 3 columnas pageSize: 8, animate: true, onLoadMore: async ({ page, pageSize }) => { await new Promise(r => setTimeout(r, 400)); const items = Array.from({ length: pageSize }, (_, i) => ({ id: (page - 1) * pageSize + i, name: `Producto ${(page - 1) * pageSize + i + 1}`, price: (Math.random() * 500 + 10).toFixed(2), })); return { items, hasMore: page < 4 }; }, renderItem: (item) => { const el = document.createElement('div'); el.className = 'inf-card'; el.innerHTML = `<strong>${item.name}</strong>$${item.price}`; return el; }, onLoad: (e) => console.log('grid page', e.detail.page), onEnd: (e) => console.log('grid done:', e.detail.total), }); // Reset to page 1 / Resetear a página 1 grid.reset();
