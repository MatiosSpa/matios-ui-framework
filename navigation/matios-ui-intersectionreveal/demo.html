MTS.IntersectionReveal

Animates elements as they enter the viewport — fade, slide, zoom, flip with stagger.

FadeSolo opacidad
fade-up

fade-right

<div class="card">Card 1</div> <div class="card">Card 2</div> <div class="card">Card 3</div> <script> new MTS.IntersectionReveal('.card', { animation: 'fade-up', duration: 600, stagger: 80, // 80ms between each / 80ms entre cada uno // Fires when element is revealed / Se dispara al revelar un elemento onReveal: (e) => { console.log(e.detail.element); // → HTMLElement console.log(e.detail.index); // → 0, 1, 2... }, }); </script>
// Each card with its own animation / Cada tarjeta con su animación document.querySelectorAll('.anim-card').forEach((card, i) => { new MTS.IntersectionReveal(card, { animation: card.dataset.anim, duration: 600, delay: i * 60, // stagger manually / escalonado manual once: true, // Fires when revealed / Se dispara al revelar onReveal: (e) => { console.log('revealed:', e.detail.index, e.detail.element); }, }); }); // Or animate all at once with stagger / O animar todos a la vez con stagger new MTS.IntersectionReveal('.card', { animation: 'fade-up', duration: 600, stagger: 80, // ms between each element / ms entre cada elemento threshold: 0.1, // trigger at 10% visibility / activar al 10% de visibilidad once: true, // animate only once / animar solo una vez onReveal: (e) => console.log(e.detail.index), });

// Stagger — cascading reveal / Revelado en cascada new MTS.IntersectionReveal('.stagger-card', { animation: 'fade-up', duration: 500, // Each element delayed by 80ms more than the previous // Cada elemento con 80ms más de delay que el anterior stagger: 80, threshold: 0.1, once: true, onReveal: (e) => console.log('revealed:', e.detail.index), }); // With base delay / Con delay base new MTS.IntersectionReveal('.hero-elements', { animation: 'zoom', duration: 700, delay: 300, // wait 300ms before starting / esperar 300ms antes stagger: 100, easing: 'cubic-bezier(.34,1.56,.64,1)', }); // Repeat on scroll / Repetir en scroll new MTS.IntersectionReveal('.stat-card', { animation: 'fade-left', once: false, // re-animate every time / re-animar cada vez threshold: 0.3, });
