MTS.TabBar

Mobile-style bottom navigation — default, pill and floating variants.

1 — Default
— cambia de tab —
<div id="app-tabbar" data-active="home"></div> <script> new MTS.TabBar('#app-tabbar', { tabs: [ { id: 'home', label: 'Home', icon: ICON_HOME }, { id: 'search', label: 'Search', icon: ICON_SEARCH }, { id: 'inbox', label: 'Inbox', icon: ICON_INBOX, badge: 3 }, { id: 'profile', label: 'Profile', icon: ICON_USER }, ], // Fires when active tab changes / Se dispara al cambiar el tab activo onChange: (e) => console.log(e.detail.id), }); </script>
new MTS.TabBar('#app-tabbar', { variant: 'default', active: 'home', showLabels: true, tabs: [ { id: 'home', label: 'Home', icon: ICON_HOME }, { id: 'search', label: 'Search', icon: ICON_SEARCH }, { id: 'inbox', label: 'Inbox', icon: ICON_INBOX, badge: 3 }, { id: 'profile', label: 'Profile', icon: ICON_USER }, ], // Fires when active tab changes / Se dispara al cambiar el tab activo onChange: (e) => { console.log(e.detail.id); // → 'search' console.log(e.detail.tab); // → { id, label, icon, badge } }, });

— usa los botones —
<!-- showLabels: false — icon only / solo ícono --> <div id="my-tabbar"></div> <script> const tb = new MTS.TabBar('#my-tabbar', { showLabels: false, tabs: [...], }); tb.setBadge('inbox', 5) // set badge / establecer badge tb.setBadge('inbox', null) // clear / limpiar tb.setActive('profile') // programmatic activation / activación programática </script>
const tb = new MTS.TabBar('#my-tabbar', { // Hide labels — icon only / Ocultar labels — solo ícono showLabels: false, variant: 'pill', tabs: TABS, onChange: (e) => console.log(e.detail.id), }); // Update badge dynamically / Actualizar badge dinámicamente tb.setBadge('inbox', 5) // show badge 5 tb.setBadge('inbox', '!') // show badge ! tb.setBadge('inbox', null) // clear badge / limpiar badge // Activate tab programmatically / Activar tab programáticamente tb.setActive('profile')
