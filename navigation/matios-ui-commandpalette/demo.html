MTS.CommandPalette

⌘K command palette — keyboard navigation, groups, icons, shortcuts, async search.

1 — Demo interactivo / Interactive demo
— abre la paleta y selecciona un comando —
<!-- MTS.CommandPalette has no HTML container — it appends to body MTS.CommandPalette no tiene contenedor HTML — se agrega al body --> <button onclick="cp.open()">Open palette</button> <!-- or press ⌘K / Ctrl+K --> <script> const cp = new MTS.CommandPalette({ commands: [...], // Fires when a command is selected / Se dispara al seleccionar un comando onSelect: (e) => { console.log(e.detail.id); // → 'create-file' console.log(e.detail.command); // → { id, label, ... } }, }); </script>
const cp = new MTS.CommandPalette({ placeholder: 'Search commands...', commands: [ // Group: Navigation / Grupo: Navegación { group: 'Navigation' }, { id: 'go-dashboard', label: 'Go to Dashboard', shortcut: '⌘D', action: () => alert('Dashboard') }, { id: 'go-settings', label: 'Open Settings', shortcut: '⌘,', action: () => alert('Settings') }, { id: 'go-profile', label: 'View Profile', action: () => alert('Profile') }, // Group: Actions / Grupo: Acciones { group: 'Actions' }, { id: 'create', label: 'Create new file', shortcut: '⌘N', action: () => alert('Create') }, { id: 'save', label: 'Save document', shortcut: '⌘S', action: () => alert('Save') }, { id: 'export', label: 'Export as PDF', action: () => alert('Export') }, { id: 'delete', label: 'Delete selected', disabled: true }, // Group: Theme / Grupo: Tema { group: 'Theme' }, { id: 'theme-dark', label: 'Switch to Dark mode', action: () => setTheme('dark') }, { id: 'theme-light', label: 'Switch to Light mode', action: () => setTheme('light') }, ], // Fires when palette opens / Se dispara al abrir onOpen: () => console.log('opened'), // Fires when palette closes / Se dispara al cerrar onClose: () => console.log('closed'), // Fires when a command is selected / Se dispara al seleccionar un comando onSelect: (e) => { console.log(e.detail.id); // → 'create' console.log(e.detail.command); // → full command object }, });
2 — Búsqueda async / Async search
— escribe algo en la paleta para simular búsqueda async —
<!-- onSearch replaces local search / onSearch reemplaza la búsqueda local --> <script> new MTS.CommandPalette({ // onSearch receives query, must return array or Promise // onSearch recibe query, debe retornar arreglo o Promesa onSearch: async (e) => { const query = e.detail.query; if (!query) return defaultCommands; const res = await fetch('/api/commands?q=' + query); return await res.json(); }, onSelect: (e) => console.log(e.detail.id), }); </script>
const DB = [ { id:'user-1', label:'Ana García', description:'ana@empresa.com', group:'Usuarios' }, { id:'user-2', label:'Pedro Martínez', description:'pedro@empresa.com', group:'Usuarios' }, { id:'user-3', label:'Laura Sánchez', description:'laura@empresa.com', group:'Usuarios' }, { id:'doc-1', label:'Reporte Q1', description:'Excel · 2.4MB', group:'Documentos' }, { id:'doc-2', label:'Contrato 2025', description:'PDF · 856KB', group:'Documentos' }, { id:'doc-3', label:'Manual de uso', description:'PDF · 1.1MB', group:'Documentos' }, ]; new MTS.CommandPalette({ placeholder: 'Buscar usuarios y documentos...', hotkey: 'j', // opens with ⌘J / abre con ⌘J // Async search — simulates API call with 200ms delay // Búsqueda async — simula llamada API con 200ms de delay onSearch: async (e) => { const query = e.detail.query; await new Promise(r => setTimeout(r, 200)); // simulate network if (!query) return DB; const q = query.toLowerCase(); return DB.filter(c => c.label.toLowerCase().includes(q) || c.description.toLowerCase().includes(q) ); }, onSelect: (e) => console.log(e.detail.id), });
