MTS.Dropdown

Menú desplegable con grupos, íconos, shortcuts, submenús y eventos.

1 — Básico con eventos
— selecciona un item —
new MTS.Dropdown('#mi-btn', {
  position: 'bottom-start',   // 'bottom-start'|'bottom-end'|'top-start'|'top-end'
  trigger:  'click',           // 'click'|'hover'
  items: [
    { label:'Ver detalle', onClick:(item) => console.log('ver', item) },
    { label:'Editar',      onClick:(item) => console.log('editar') },
    { divider: true },
    { label:'Eliminar', danger:true, onClick:(item) => console.log('eliminar') },
  ],
  onSelect: (e) => console.log('seleccionado:', e.detail.item.label),
  onOpen:   (e) => console.log('abierto'),
  onClose:  (e) => console.log('cerrado'),
})
2 — Con íconos, shortcuts y submenús
— selecciona un item —
new MTS.Dropdown('#btn', {
  items: [
    { label:'Ver',      icon:MTS.Icon.get('eye'),    shortcut:'⌘D' },
    { label:'Editar',   icon:MTS.Icon.get('edit-2'), shortcut:'⌘E' },
    { label:'Mover a',  icon:MTS.Icon.get('folder'),
      items: [                    // submenú anidado
        { label:'Mis documentos', icon:MTS.Icon.get('folder') },
        { label:'Compartidos',    icon:MTS.Icon.get('users')  },
      ]},
    { divider:true },
    { label:'Eliminar', icon:MTS.Icon.get('trash'), danger:true },
  ],
  onSelect: (e) => console.log('item:', e.detail.item.label),
})

// trigger:'hover' — abre al pasar el mouse
new MTS.Dropdown('#btn', { trigger:'hover', items:[...] })
3 — Con grupos
— selecciona una vista —
new MTS.Dropdown('#btn', {
  items: [
    { group:'Diseño' },
    { label:'Lista',      icon:MTS.Icon.get('list')         },
    { label:'Cuadrícula', icon:MTS.Icon.get('grid')         },
    { divider:true },
    { group:'Avanzado' },
    { label:'Tabla',      icon:MTS.Icon.get('table')        },
    { label:'Kanban',     icon:MTS.Icon.get('columns')      },
  ],
})
4 — Submenús anidados (hasta 4 niveles)
— navega por los submenús —
new MTS.Dropdown('#btn', {
  items: [
    { label:'Nivel 1 — A',
      items: [
        { label:'Nivel 2 — A1',
          items: [
            { label:'Nivel 3 — final', onClick:(item) => console.log(item.label) },
          ]},
      ]},
  ],
  onSelect: (e) => console.log('seleccionado:', e.detail.item.label),
})
