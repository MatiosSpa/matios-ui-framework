MTS.Tabs

Pestañas con variantes underline, pill, card y bordered — horizontal y vertical.

1 — Underline (default)
— cambia de pestaña —
new MTS.Tabs('#el', {
  variant: 'underline',
  active:  'general',
  tabs: [
    { id:'general',   label:'General',   content:'...' },
    { id:'seguridad', label:'Seguridad', content:'...' },
    { id:'notif',     label:'Notif.',    content:'...', badge:4 },
    { id:'api',       label:'API',       content:'...', disabled:true },
  ],
  onChange: (e) => console.log('tab activa:', e.detail.id, e.detail.tab.label),
})
2 — Bordered — caja con bordes (estilo acordeón)
— cambia de pestaña —
// variante 'bordered': nav + panel forman una caja cerrada
// tab activa → borde superior colored + panel con bordes completos
new MTS.Tabs('#el', {
  variant: 'bordered',
  tabs: [
    { id:'resumen', label:'Resumen',    content:'...' },
    { id:'detalle', label:'Detalle',    content:'...' },
    { id:'historial',label:'Historial', content:'...' },
  ],
  onChange: (e) => console.log('tab:', e.detail.id),
})
3 — Pill
new MTS.Tabs('#el', {
  variant: 'pill',
  tabs: [
    { id:'lista',      label:'Lista'      },
    { id:'cuadricula', label:'Cuadrícula' },
    { id:'mapa',       label:'Mapa'       },
  ],
})
4 — Card horizontal
new MTS.Tabs('#el', {
  variant: 'card',
  tabs: [...],
})
5 — Vertical bordered
— cambia de pestaña —
new MTS.Tabs('#el', {
  variant:   'bordered',
  direction: 'vertical',
  navWidth:  '160px',
  stretch:    true,
  tabs: [...]
})
6 — Con íconos y badges
new MTS.Tabs('#el', {
  variant: 'bordered',
  tabs: [
    { id:'inbox', label:'Inbox',    icon:MTS.Icon.get('inbox'),  badge:12 },
    { id:'sent',  label:'Enviados', icon:MTS.Icon.get('send')             },
    { id:'draft', label:'Borrador', icon:MTS.Icon.get('edit-2'), badge:3  },
    { id:'trash', label:'Papelera', icon:MTS.Icon.get('trash'),  disabled:true },
  ],
})
7 — API — setActive() programático
— cambia de pestaña —
const tabs = new MTS.Tabs('#el', {
  variant: 'underline',
  tabs: [...],
  onChange: (e) => {
    console.log('id:',    e.detail.id)
    console.log('tab:',   e.detail.tab.label)
  },
})
tabs.setActive('mi-tab')   // programático
tabs.addTab({ id:'nuevo', label:'Nuevo', content:'...' })
tabs.removeTab('viejo')
