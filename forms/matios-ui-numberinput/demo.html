MTS.NumberInput

Input numérico con botones +/−, min/max, step y formatos de moneda y porcentaje.

1 — Básico con min/max/step
— interactúa con los inputs —
<div id="inp-qty" data-label="Quantity" data-min="0" data-max="100" data-step="1"> </div> <script> new MTS.NumberInput('#inp-qty', { value: 1, onChange: (value, formatted) => console.log(value), }); </script>
// Básico con límites new MTS.NumberInput('#el', { label: 'Cantidad', value: 1, min: 0, max: 10, onChange: (val, formatted) => console.log(val), }) // Con step personalizado new MTS.NumberInput('#el', { label: 'Cada 5 unidades', value: 0, min: 0, max: 100, step: 5, }) // Con decimales new MTS.NumberInput('#el', { label: 'Con 2 decimales', value: 1.5, step: 0.1, decimals: 2, })
2 — Formatos: moneda y porcentaje
— cambia los valores —
<div id="inp-price" data-label="Price" data-min="0" data-step="500"> </div> <script> new MTS.NumberInput('#inp-price', { format: 'currency', currency: 'CLP', locale: 'es-CL', value: 15000, onChange: (value, formatted) => console.log(formatted), }); </script>
// Moneda CLP new MTS.NumberInput('#el', { label: 'Precio (CLP)', value: 15000, format: 'currency', currency: 'CLP', locale: 'es-CL', step: 500, min: 0, }) // Moneda USD new MTS.NumberInput('#el', { label: 'Precio (USD)', value: 99, format: 'currency', currency: 'USD', locale: 'en-US', step: 1, decimals: 2, min: 0, }) // Porcentaje new MTS.NumberInput('#el', { label: 'Descuento', value: 10, format: 'percent', min: 0, max: 100, step: 5, })
3 — Prefix y Suffix
<div id="inp-prefix" data-label="Price USD" data-prefix="$" data-suffix="USD" data-decimals="2"> </div> <script> new MTS.NumberInput('#inp-prefix', { value: 99.99 }); </script>
new MTS.NumberInput('#el', { label:'Monto', prefix:'$', value:500, step:100 }) new MTS.NumberInput('#el', { label:'Peso', suffix:'kg', value:75, step:0.5, decimals:1 }) new MTS.NumberInput('#el', { label:'Temperatura', prefix:'~', suffix:'°C', value:20, min:-20, max:50 })
4 — Tamaños
<div id="inp-sm" data-size="sm" data-label="Small"></div> <div id="inp-md" data-size="md" data-label="Medium"></div> <div id="inp-lg" data-size="lg" data-label="Large"></div> <script> new MTS.NumberInput('#inp-sm', { value: 10 }); new MTS.NumberInput('#inp-md', { value: 10 }); new MTS.NumberInput('#inp-lg', { value: 10 }); </script>
new MTS.NumberInput('#el', { label:'SM', size:'sm', value:1 }) new MTS.NumberInput('#el', { label:'MD', size:'md', value:1 }) // default new MTS.NumberInput('#el', { label:'LG', size:'lg', value:1 })
5 — API — setValue · setError · disable · enable
— usa los botones —
<div id="inp-api" data-label="Controlled input" data-min="0" data-max="100"> </div> <script> const ni = new MTS.NumberInput('#inp-api', { value: 50, onChange: (value, formatted) => console.log(value, formatted), }); ni.getValue() ni.setValue(75) ni.setError('Out of range') ni.clearError() ni.disable() ni.enable() </script>
const ni = new MTS.NumberInput('#el', { label: 'Puntos', value: 50, min: 0, max: 100, onChange: (val, formatted) => console.log(val, formatted), }) ni.getValue() // → 50 ni.setValue(75) // sin disparar onChange ni.setValue(75, true) // silent — no dispara onChange ni.setError('Fuera de rango') ni.clearError() ni.disable() ni.enable()

— cambia el valor —
