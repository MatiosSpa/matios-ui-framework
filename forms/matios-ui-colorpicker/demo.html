MTS.ColorPicker

Selector de color con sliders HSL, paleta de presets y formatos hex/rgb/hsl.

1 — Trigger (popup) — tamaños
— elige un color —
new MTS.ColorPicker('#el', {
  label:   'Color principal',
  value:   '#4f8eff',
  format:  'hex',  // 'hex' | 'rgb' | 'hsl'
  onChange: ({ hex, value }) => {
    console.log(hex, value)
  },
})
2 — Inline (siempre visible)
— mueve los sliders —
new MTS.ColorPicker('#el', {
  inline:  true,
  value:   '#34d399',
  onChange: ({ hex, value }) => {},
})
3 — Formatos hex · rgb · hsl + API
new MTS.ColorPicker('#el', { format:'hex' })  // → '#4f8eff'
new MTS.ColorPicker('#el', { format:'rgb' })  // → 'rgb(79, 142, 255)'
new MTS.ColorPicker('#el', { format:'hsl' })  // → 'hsl(221, 100%, 65%)'

const cp = new MTS.ColorPicker('#el', { value:'#4f8eff' })
cp.getValue()           // según format actual
cp.getHex()             // siempre hex
cp.setValue('#ff0000')
cp.setFormat('rgb')
cp.open() / cp.close()
4 — Sin sliders · sin presets · disabled
new MTS.ColorPicker('#el', { showSliders:false, label:'Sin sliders' })
new MTS.ColorPicker('#el', { showPresets:false, label:'Sin presets' })
new MTS.ColorPicker('#el', { disabled:true,     label:'Deshabilitado', value:'#818cf8' })
HTML declarativo
— selecciona color —
<div id="miColor" data-value="#7c3aed" data-label="Color de marca" data-format="hex"></div>
<script>new MTS.ColorPicker('#miColor', { onChange: ({ hex }) => console.log(hex) })</script>
