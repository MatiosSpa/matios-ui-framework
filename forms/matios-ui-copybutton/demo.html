MTS.CopyButton

Copy-to-clipboard button with automatic visual feedback and target element support.

1 — Variantes y tamaños / Variants and sizes
— haz click para copiar —
<button id="btn-copy" data-text="npm install matios-ui" data-label="Copy" data-variant="secondary"> </button> <script> new MTS.CopyButton('#btn-copy', { // Fires after successful copy / Se dispara tras copiar onCopy: (text) => console.log('copied:', text), }); </script>
new MTS.CopyButton('#btn-primary', { text: 'npm install matios-ui', variant: 'primary', onCopy: (text) => console.log('copied:', text), }); new MTS.CopyButton('#btn-secondary', { text: 'npm install matios-ui', variant: 'secondary', }); // Ghost variant / Variante ghost new MTS.CopyButton('#btn-ghost', { text: 'npm install matios-ui', variant: 'ghost', }); // Sizes / Tamaños new MTS.CopyButton('#btn-sm', { text: 'copy me', size: 'sm', variant: 'secondary' }); new MTS.CopyButton('#btn-lg', { text: 'copy me', size: 'lg', variant: 'secondary' });
2 — Solo ícono / Icon only
<button id="btn-icon" data-text="text to copy" data-icon-only data-variant="ghost"> </button> <script> new MTS.CopyButton('#btn-icon'); </script>
// Icon only — no label / Solo ícono — sin label new MTS.CopyButton('#btn-sm', { text: 'copy me', iconOnly: true, size: 'sm', variant: 'ghost' }); new MTS.CopyButton('#btn-md', { text: 'copy me', iconOnly: true, size: '', variant: 'secondary' }); new MTS.CopyButton('#btn-pri', { text: 'copy me', iconOnly: true, size: '', variant: 'primary' });

new MTS.CopyButton('#btn-apikey', { // Reads .value from target element / Lee .value del elemento objetivo target: '#api-key-input', iconOnly: true, variant: 'primary', onCopy: (text) => console.log('copied:', text), }); new MTS.CopyButton('#btn-install', { target: '#install-input', iconOnly: true, variant: 'secondary', });
4 — Bloque de código / Code block
npm install matios-ui
<div style="position:relative"> <pre id="code-block">npm install matios-ui</pre> <button id="btn-copy" style="position:absolute;top:6px;right:6px"> </button> </div> <script> new MTS.CopyButton('#btn-copy', { target: '#code-block', // reads textContent / lee textContent iconOnly: true, size: 'sm', variant: 'ghost', }); </script>
new MTS.CopyButton('#btn-code', { // Reads .textContent from <pre> / Lee .textContent del <pre> target: '#code-block', iconOnly: true, size: 'sm', variant: 'ghost', });
5 — API y callback / API and callback
— usa los botones —
<button id="btn-copy" data-text="initial text" data-variant="secondary"> </button> <script> const btn = new MTS.CopyButton('#btn-copy', { onCopy: (text) => console.log('copied:', text), }); btn.setText('new text') // change text at runtime btn.copy() // copy programmatically </script>
const btn = new MTS.CopyButton('#btn-copy', { text: 'initial text to copy', variant: 'secondary', // Fires after copy / Se dispara tras copiar onCopy: (text) => { console.log('copied:', text); }, }); // Change text at runtime / Cambiar texto en runtime btn.setText('new text to copy') // Copy programmatically / Copiar programáticamente btn.copy()
