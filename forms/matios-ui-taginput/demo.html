MTS.TagInput

Input de etiquetas con sugerencias locales y autocomplete async.

1 — Sugerencias locales / Local suggestions
— agrega o elimina etiquetas —
<div id="tags-project"></div> <script> new MTS.TagInput('#tags-project', { label: 'Etiquetas', tags: [ { uid: 'urgent', name: 'Urgente' }, { uid: 'review', name: 'Revisión' }, ], suggestions: [ { uid: 'urgent', name: 'Urgente' }, { uid: 'pending', name: 'Pendiente' }, { uid: 'approved', name: 'Aprobado' }, { uid: 'blocked', name: 'Bloqueado' }, { uid: 'review', name: 'Revisión' }, { uid: 'in-progress', name: 'En progreso' }, ], onChange: (e) => console.log(e.detail.tags), // → [{ uid: 'urgent', name: 'Urgente' }, ...] }); </script>
const tags = new MTS.TagInput('#tags-project', { label: 'Etiquetas', // Tags iniciales — { uid, name } tags: [ { uid: 'urgent', name: 'Urgente' }, { uid: 'review', name: 'Revisión' }, ], // Sugerencias locales — { uid, name } suggestions: [ { uid: 'urgent', name: 'Urgente' }, { uid: 'pending', name: 'Pendiente' }, { uid: 'approved', name: 'Aprobado' }, { uid: 'blocked', name: 'Bloqueado' }, { uid: 'review', name: 'Revisión' }, { uid: 'in-progress', name: 'En progreso' }, ], onChange: (e) => console.log(e.detail.tags), // → [{ uid: 'urgent', name: 'Urgente' }, { uid: 'review', name: 'Revisión' }] onAdd: (e) => console.log('added:', e.detail.tag), // { uid, name } onRemove: (e) => console.log('removed:', e.detail.tag), // { uid, name } }); // API tags.getTags() // → [{ uid: 'urgent', name: 'Urgente' }, ...] tags.addTag({ uid: 'critical', name: 'Crítico' }) tags.removeTag('review') // elimina por uid tags.setTags([{ uid: 'new', name: 'Nuevo' }])
2 — Autocomplete async

💡 Escribe: java, vue, python, docker

<div id="tags-stack"></div> <script> // onSearch retorna [{ uid, name }] directamente — no necesita setSuggestions new MTS.TagInput('#tags-stack', { label: 'Tech stack', allowCustom: false, debounce: 300, onSearch: async (query) => { const TECHS = [ { uid: 'js', name: 'JavaScript' }, { uid: 'ts', name: 'TypeScript' }, { uid: 'py', name: 'Python' }, { uid: 'vue', name: 'Vue' }, { uid: 'rct', name: 'React' }, { uid: 'docker', name: 'Docker' }, ]; await new Promise(r => setTimeout(r, 200)); return TECHS.filter(t => t.name.toLowerCase().includes(query.toLowerCase())); }, onChange: (e) => console.log(e.detail.tags), // → [{ uid: 'js', name: 'JavaScript' }, ...] }); </script>
new MTS.TagInput('#tags-stack', { label: 'Tech stack', tags: [{ uid: 'js', name: 'JavaScript' }], allowCustom: false, debounce: 300, // onSearch retorna [{ uid, name }] — se aplica automáticamente onSearch: async (query) => { // Simula llamada a API / Simulates API call const res = await fetch('/api/users?q=' + encodeURIComponent(query)); return res.json(); // → [{ uid, name }, ...] // O con datos locales / Or with local data: // const PEOPLE = [ // { uid: 'ana@mail.com', name: 'Ana López' }, // { uid: 'carlos@mail.com', name: 'Carlos Ruiz' }, // ]; // return PEOPLE.filter(p => p.name.toLowerCase().includes(query.toLowerCase())); }, onChange: (e) => console.log(e.detail.tags), // → [{ uid: 'ana@mail.com', name: 'Ana López' }, ...] });
