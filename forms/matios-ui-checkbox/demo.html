MTS.Checkbox

Checkbox individual e indeterminado, grupos vertical y horizontal, estados disabled.

1 — Checkbox individual
Normal
Pre-marcado
Indeterminado
Disabled
Disabled checked
— interactúa con los checkboxes —
<!-- Checkbox individual --> <div class="mts-checkbox-wrap"> <input type="checkbox" class="mts-checkbox"> <label class="mts-checkbox__label">Acepto los términos</label> </div> <!-- Indeterminado --> <div class="mts-checkbox-wrap"> <input type="checkbox" class="mts-checkbox" id="chk-ind"> <label class="mts-checkbox__label">Selección parcial</label> </div> <script>document.getElementById('chk-ind').indeterminate = true;</script> <!-- Disabled --> <div class="mts-checkbox-wrap"> <input type="checkbox" class="mts-checkbox" disabled> <label class="mts-checkbox__label">No disponible</label> </div>
// Normal const chk = new MTS.Checkbox('#el', { label: 'Acepto los términos', checked: false, onChange: (e) => console.log(e.detail.checked), }); // Indeterminado const chkInd = new MTS.Checkbox('#el', { label: 'Selección parcial', indeterminate: true, }); // Disabled checked const chkDis = new MTS.Checkbox('#el', { label: 'No disponible', checked: true, disabled: true, }); // API chk.isChecked() // → boolean chk.setChecked(true) chk.setIndeterminate(true) chk.toggle()
2 — Grupo vertical (default)
— selecciona opciones —
<div class="mts-checkbox-group"> <div class="mts-checkbox-wrap"> <input type="checkbox" class="mts-checkbox" value="pdf" checked> <label class="mts-checkbox__label">PDF</label> </div> <div class="mts-checkbox-wrap"> <input type="checkbox" class="mts-checkbox" value="xlsx"> <label class="mts-checkbox__label">Excel</label> </div> <div class="mts-checkbox-wrap"> <input type="checkbox" class="mts-checkbox" value="csv" disabled> <label class="mts-checkbox__label">CSV (no disponible)</label> </div> </div>
const group = new MTS.CheckboxGroup('#el', { options: [ { value: 'pdf', label: 'PDF' }, { value: 'xlsx', label: 'Excel' }, { value: 'csv', label: 'CSV', disabled: true }, { value: 'docx', label: 'Word' }, ], value: ['pdf'], onChange: (e) => console.log(e.detail.value), }); group.getValue() // → ['pdf', 'xlsx']
3 — Grupo horizontal
<!-- Agregar clase --horizontal para layout en fila --> <div class="mts-checkbox-group mts-checkbox-group--horizontal"> <div class="mts-checkbox-wrap"> <input type="checkbox" class="mts-checkbox" value="lunes" checked> <label class="mts-checkbox__label">Lunes</label> </div> <div class="mts-checkbox-wrap"> <input type="checkbox" class="mts-checkbox" value="martes"> <label class="mts-checkbox__label">Martes</label> </div> <!-- ... --> </div>
const group = new MTS.CheckboxGroup('#el', { horizontal: true, // ← activa layout en fila options: [ { value: 'lunes', label: 'Lunes' }, { value: 'martes', label: 'Martes' }, { value: 'miercoles', label: 'Miércoles' }, { value: 'jueves', label: 'Jueves' }, { value: 'viernes', label: 'Viernes' }, ], value: ['lunes', 'miercoles'], });
4 — Grupo completamente deshabilitado
<!-- disabled en cada input individualmente --> <div class="mts-checkbox-group"> <div class="mts-checkbox-wrap"> <input type="checkbox" class="mts-checkbox" value="a" checked disabled> <label class="mts-checkbox__label">Opción A</label> </div> <div class="mts-checkbox-wrap"> <input type="checkbox" class="mts-checkbox" value="b" disabled> <label class="mts-checkbox__label">Opción B</label> </div> </div>
// disabled: true deshabilita todo el grupo const group = new MTS.CheckboxGroup('#el', { disabled: true, options: [ { value: 'a', label: 'Opción A' }, { value: 'b', label: 'Opción B' }, { value: 'c', label: 'Opción C' }, ], value: ['a'], });
