MTS.ConfirmButton

Two-step inline confirmation — prevents accidental actions without a modal.

1 — Variantes / Variants
— interactúa con los botones —
<div id="btn-delete" data-label="Delete record" data-confirm-label="Yes, delete" data-cancel-label="Cancel" data-variant="secondary" data-confirm-variant="danger" data-timeout="4000"> </div> <script> new MTS.ConfirmButton('#btn-delete', { onConfirm: () => console.log('deleted'), onCancel: () => console.log('cancelled'), }); </script>
// Danger variant / Variante danger new MTS.ConfirmButton('#btn-delete', { label: 'Delete record', confirmLabel: 'Yes, delete', cancelLabel: 'Cancel', variant: 'secondary', confirmVariant: 'danger', timeout: 4000, // Fires when user confirms / Se dispara al confirmar onConfirm: () => console.log('deleted'), // Fires on cancel or timeout / Se dispara al cancelar o expirar onCancel: () => console.log('cancelled'), }); // Warning variant new MTS.ConfirmButton('#btn-archive', { label: 'Archive', confirmLabel: 'Yes, archive', variant: 'secondary', confirmVariant: 'warning', onConfirm: () => console.log('archived'), });
2 — Sin timeout / Without timeout
<!-- timeout: 0 disables auto-cancel / timeout: 0 desactiva el auto-cancelar --> <div id="btn-no-timeout" data-label="Logout" data-confirm-label="Yes, logout" data-variant="ghost" data-confirm-variant="danger" data-timeout="0"> </div> <script> new MTS.ConfirmButton('#btn-no-timeout', { onConfirm: () => console.log('logged out'), }); </script>
// No timeout — stays open until user decides // Sin timeout — permanece abierto hasta que el usuario decide new MTS.ConfirmButton('#btn-logout', { label: 'Logout', confirmLabel: 'Yes, logout', variant: 'ghost', confirmVariant: 'danger', timeout: 0, // 0 = no auto-cancel / 0 = sin auto-cancelar onConfirm: () => logout(), });
3 — Tamaños / Sizes
<div id="btn-sm" data-label="Delete" data-size="sm" data-confirm-variant="danger"></div> <div id="btn-md" data-label="Delete" data-size="md" data-confirm-variant="danger"></div> <div id="btn-lg" data-label="Delete" data-size="lg" data-confirm-variant="danger"></div> <script> ['sm','md','lg'].forEach(s => new MTS.ConfirmButton('#btn-' + s, { onConfirm: () => {} }) ); </script>
// Size: 'sm' | 'md' | 'lg' / Tamaño new MTS.ConfirmButton('#btn-sm', { label: 'Delete', size: 'sm', confirmVariant: 'danger', onConfirm: () => {} }); new MTS.ConfirmButton('#btn-md', { label: 'Delete', size: 'md', confirmVariant: 'danger', onConfirm: () => {} }); new MTS.ConfirmButton('#btn-lg', { label: 'Delete', size: 'lg', confirmVariant: 'danger', onConfirm: () => {} });
4 — API
<div id="btn-api" data-label="Delete" data-confirm-variant="danger"></div> <script> const btn = new MTS.ConfirmButton('#btn-api', { onConfirm: () => console.log('confirmed'), }); btn.reset() // reset to initial state / resetear al estado inicial btn.disable() // disable / deshabilitar btn.enable() // enable / habilitar </script>
const btn = new MTS.ConfirmButton('#btn-api', { label: 'Delete', confirmVariant: 'danger', onConfirm: () => console.log('confirmed'), onCancel: () => console.log('cancelled'), }); // Reset to initial state / Resetear al estado inicial btn.reset() // Disable all buttons / Deshabilitar todos los botones btn.disable() // Enable all buttons / Habilitar todos los botones btn.enable()
