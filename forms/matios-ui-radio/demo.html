MTS.Radio

Grupo de radio buttons — vertical, horizontal, con opción deshabilitada y grupo completo disabled.

1 — Vertical (default)
— selecciona una opción —
<div class="mts-radio-group"> <div class="mts-radio-wrap"> <input type="radio" class="mts-radio" name="prioridad" value="high"> <label class="mts-radio__label">Alta</label> </div> <div class="mts-radio-wrap"> <input type="radio" class="mts-radio" name="prioridad" value="medium" checked> <label class="mts-radio__label">Media</label> </div> <div class="mts-radio-wrap"> <input type="radio" class="mts-radio" name="prioridad" value="low"> <label class="mts-radio__label">Baja</label> </div> </div>
const radio = new MTS.Radio('#el', { name: 'prioridad', value: 'medium', options: [ { value: 'high', label: 'Alta' }, { value: 'medium', label: 'Media' }, { value: 'low', label: 'Baja' }, ], onChange: (e) => console.log(e.detail.value), }); radio.getValue() // → 'medium' radio.setValue('high')
2 — Horizontal
— selecciona una opción —
<!-- Agregar clase --horizontal para layout en fila --> <div class="mts-radio-group mts-radio-group--horizontal"> <div class="mts-radio-wrap"> <input type="radio" class="mts-radio" name="vista" value="tabla" checked> <label class="mts-radio__label">Tabla</label> </div> <div class="mts-radio-wrap"> <input type="radio" class="mts-radio" name="vista" value="tarjetas"> <label class="mts-radio__label">Tarjetas</label> </div> <div class="mts-radio-wrap"> <input type="radio" class="mts-radio" name="vista" value="lista"> <label class="mts-radio__label">Lista</label> </div> </div>
const radio = new MTS.Radio('#el', { horizontal: true, // ← activa layout en fila name: 'vista', value: 'tabla', options: [ { value: 'tabla', label: 'Tabla' }, { value: 'tarjetas', label: 'Tarjetas' }, { value: 'lista', label: 'Lista' }, ], onChange: (e) => console.log(e.detail.value), });
3 — Con opción deshabilitada
<div class="mts-radio-group mts-radio-group--horizontal"> <div class="mts-radio-wrap"> <input type="radio" class="mts-radio" name="plan" value="free" checked> <label class="mts-radio__label">Free</label> </div> <div class="mts-radio-wrap"> <input type="radio" class="mts-radio" name="plan" value="pro"> <label class="mts-radio__label">Pro</label> </div> <!-- Opción individual disabled --> <div class="mts-radio-wrap"> <input type="radio" class="mts-radio" name="plan" value="enterprise" disabled> <label class="mts-radio__label">Enterprise (próximamente)</label> </div> </div>
const radio = new MTS.Radio('#el', { horizontal: true, name: 'plan', value: 'free', options: [ { value: 'free', label: 'Free' }, { value: 'pro', label: 'Pro' }, { value: 'enterprise', label: 'Enterprise (próximamente)', disabled: true }, ], });
4 — Grupo completo deshabilitado
<!-- disabled en cada input individualmente --> <div class="mts-radio-group"> <div class="mts-radio-wrap"> <input type="radio" class="mts-radio" name="estado" value="activo" checked disabled> <label class="mts-radio__label">Activo</label> </div> <div class="mts-radio-wrap"> <input type="radio" class="mts-radio" name="estado" value="inactivo" disabled> <label class="mts-radio__label">Inactivo</label> </div> </div>
// disabled: true deshabilita todo el grupo const radio = new MTS.Radio('#el', { disabled: true, name: 'estado', value: 'activo', options: [ { value: 'activo', label: 'Activo' }, { value: 'inactivo', label: 'Inactivo' }, { value: 'pendiente', label: 'Pendiente' }, ], });
