MTS.Input

Input de texto, email, password, número y textarea — con validación integrada.

1 — Tipos de input
new MTS.Input('#el', { type:'text',     label:'Nombre',     placeholder:'Juan Pérez', hint:'Mínimo 3 caracteres' })
new MTS.Input('#el', { type:'email',    label:'Email',      placeholder:'correo@ejemplo.com' })
new MTS.Input('#el', { type:'password', label:'Contraseña', showPassword:true })
new MTS.Input('#el', { type:'number',   label:'Cantidad',   rules:{ min:0, max:999 } })
new MTS.Input('#el', { type:'text',     label:'Disabled',   disabled:true })
// Establecer error manualmente
const inp = new MTS.Input('#el', { type:'text', label:'Con error' })
inp.setError('Este campo no es válido')
2 — Textarea
new MTS.Input('#el', {
  type:        'textarea',
  label:       'Descripción',
  placeholder: 'Escribe aquí...',
  rows:        4,
  maxLength:   200,
  showCounter: true,
})
3 — Con íconos
new MTS.Input('#el', {
  label:    'Buscar',
  iconLeft: MTS.Icon.get('search'),
})
new MTS.Input('#el', {
  label:     'Website',
  iconRight: MTS.Icon.get('external-link'),
})
new MTS.Input('#el', {
  label:     'Usuario',
  iconLeft:  MTS.Icon.get('user'),
  iconRight: MTS.Icon.get('check-circle'),
})
4 — Estados de validación
// Estado success
const inp = new MTS.Input('#el', { label:'Email verificado', value:'correo@ok.com' })
inp.setSuccess?.()    // o clase mts-input-success

// Estado error con mensaje
const inp2 = new MTS.Input('#el', { label:'Email inválido' })
inp2.setError('Ingresa un email válido')
5 — API interactiva
— escribe algo en el input —
const inp = new MTS.Input('#el', {
  label:       'Input API',
  placeholder: 'Escribe algo...',
  onChange:    (e) => console.log('value:', e.detail.value),
  onBlur:      (e) => console.log('blur'),
  onFocus:     (e) => console.log('focus'),
})

inp.getValue()          // → string
inp.setValue('texto')
inp.setError('mensaje')
inp.clearError()
inp.disable()
inp.enable()
inp.focus()
