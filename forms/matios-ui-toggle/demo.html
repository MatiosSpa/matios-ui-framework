MTS.Toggle

Switch on/off — three sizes, with or without label, disabled state.

1 — Tamaños / Sizes: sm · md · lg
sm
md (default)
lg
<!-- sm --> <div id="toggle-sm"></div> <!-- md (default) --> <div id="toggle-md"></div> <!-- lg --> <div id="toggle-lg"></div> <script> new MTS.Toggle('#toggle-sm', { label: 'Small', size: 'sm' }); new MTS.Toggle('#toggle-md', { label: 'Medium', size: 'md', checked: true }); new MTS.Toggle('#toggle-lg', { label: 'Large', size: 'lg' }); </script>
// Size variant: 'sm' | 'md' | 'lg' (default: 'md') // Variante de tamaño: 'sm' | 'md' | 'lg' (default: 'md') new MTS.Toggle('#toggle-sm', { label: 'Small', size: 'sm' }); new MTS.Toggle('#toggle-md', { label: 'Medium', size: 'md', checked: true }); new MTS.Toggle('#toggle-lg', { label: 'Large', size: 'lg' });
2 — Con label / Con eventos / Sin label
— interactúa con los toggles / interact with the toggles —
<div id="toggle-notif"></div> <div id="toggle-maint"></div> <div id="toggle-bare"></div> <script> new MTS.Toggle('#toggle-notif', { label: 'Active notifications', checked: true, onChange: (e) => console.log('checked:', e.detail.checked), }); new MTS.Toggle('#toggle-maint', { label: 'Maintenance mode', onChange: (e) => console.log('checked:', e.detail.checked), }); // Without label / Sin label new MTS.Toggle('#toggle-bare', { checked: false, onChange: (e) => console.log('checked:', e.detail.checked), }); </script>
// With label and onChange event // Con label y evento onChange const t1 = new MTS.Toggle('#toggle-notif', { // Text next to the switch / Texto junto al switch label: 'Active notifications', // Initial state / Estado inicial checked: true, // Fires when state changes / Se dispara al cambiar el estado onChange: (e) => { console.log('checked:', e.detail.checked); // → true | false }, }); // Without label / Sin label const t2 = new MTS.Toggle('#toggle-bare', { checked: false, onChange: (e) => console.log('checked:', e.detail.checked), }); // API t1.isChecked() // → boolean t1.setChecked(false) // set state / establece estado t1.toggle() // invert state / invierte estado
3 — Disabled
<div id="toggle-dis-off"></div> <div id="toggle-dis-on"></div> <script> // Disabled off / Deshabilitado apagado new MTS.Toggle('#toggle-dis-off', { label: 'Not available', disabled: true, checked: false, }); // Disabled on / Deshabilitado encendido new MTS.Toggle('#toggle-dis-on', { label: 'Forced active', disabled: true, checked: true, }); </script>
// Disabled off / Deshabilitado apagado new MTS.Toggle('#toggle-dis-off', { label: 'Not available', // Disables all interaction / Deshabilita toda interacción disabled: true, checked: false, }); // Disabled on / Deshabilitado encendido new MTS.Toggle('#toggle-dis-on', { label: 'Forced active', disabled: true, checked: true, });
4 — API programática / Programmatic API
— usa los botones para llamar la API / use the buttons to call the API —
<div id="toggle-api"></div> <script> const toggle = new MTS.Toggle('#toggle-api', { label: 'Controlled toggle', checked: false, onChange: (e) => console.log('changed:', e.detail.checked), }); toggle.isChecked() // → boolean toggle.setChecked(true) toggle.setChecked(false) toggle.toggle() </script>
const toggle = new MTS.Toggle('#toggle-api', { label: 'Controlled toggle', checked: false, onChange: (e) => console.log('changed:', e.detail.checked), }); // Returns current state / Retorna el estado actual toggle.isChecked() // → boolean // Sets state programmatically / Establece el estado programáticamente toggle.setChecked(true) toggle.setChecked(false) // Toggles current state / Invierte el estado actual toggle.toggle()
