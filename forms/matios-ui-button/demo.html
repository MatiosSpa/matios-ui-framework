5 — Grupos de botones
Selector de vista — activo al hacer click
Acciones — variantes mixtas
Tamaño sm — group con size
Paginación simple
// Selector con estado activo automático
const grupo = new MTS.ButtonGroup('#contenedor', [
  { label: 'Día',    variant: 'secondary' },
  { label: 'Semana', variant: 'secondary' },
  { label: 'Mes',    variant: 'primary'   },  // inicia activo
], { activeOnClick: true })

// Marcar activo programáticamente
grupo.setActive(2)          // activa el tercer botón
grupo.getButton(0).disable()  // deshabilita el primero

// Acciones con variantes mixtas
new MTS.ButtonGroup('#contenedor', [
  { label: 'Exportar', variant: 'ghost'  },
  { label: 'Duplicar', variant: 'ghost'  },
  { label: 'Eliminar', variant: 'danger' },
])

// Con tamaño uniforme
new MTS.ButtonGroup('#contenedor', [...botones], { size: 'sm' })

7 — Eventos onClick

Haz click en cualquier botón — el resultado aparece debajo en tiempo real.

<!-- El HTML declara el botón, el JS maneja el evento --> <button id="btn-guardar" data-variant="primary" data-label="Guardar cambios"></button> <button id="btn-contador" data-variant="secondary" data-label="Contador: 0"></button> <script> // onClick en la instancia — forma recomendada const btnGuardar = new MTS.Button('#btn-guardar', { onClick: (event, instance) => { instance.setLoading(true); instance.setLabel('Guardando...'); setTimeout(() => { instance.setLoading(false); instance.setLabel('✓ Guardado'); }, 1500); }, }); // .on() — permite múltiples listeners sin reemplazar el onClick let clicks = 0; const btnContador = new MTS.Button('#btn-contador'); btnContador.on('click', (e, inst) => { clicks++; inst.setLabel('Contador: ' + clicks); }); // Evento DOM nativo — útil cuando no tienes la instancia document.getElementById('btn-guardar') .addEventListener('mts:button:click', (e) => { console.log('instancia:', e.detail.button.label); }); </script>
// onClick en el constructor — forma más directa const btn = new MTS.Button('#btn', { label: 'Guardar cambios', variant: 'primary', onClick: (event, instance) => { instance.setLoading(true); instance.setLabel('Guardando...'); setTimeout(() => { instance.setLoading(false); instance.setLabel('✓ Guardado'); instance.setVariant('secondary'); }, 1500); }, }); // .on() — agrega listeners sin reemplazar el onClick original btn.on('click', (event, instance) => { console.log('click en:', instance.label); }); // .off() — remueve un listener específico const handler = (e, i) => console.log('click'); btn.on('click', handler); btn.off('click', handler); // Evento DOM nativo — útil cuando no tienes la instancia document.getElementById('btn') .addEventListener('mts:button:click', (e) => { console.log('instancia desde DOM:', e.detail.button); });
