3 — Con íconos
Ícono izquierdo
Ícono derecho
Solo ícono (iconOnly)
// Ícono izquierdo
new MTS.Button('#btn', {
  label:    'Guardar',
  variant:  'primary',
  iconLeft: MTS.Icon.get('save'),
})

// Ícono derecho (chevron)
new MTS.Button('#btn', {
  label:     'Siguiente',
  variant:   'secondary',
  iconRight: MTS.Icon.get('arrow-right'),
})

// Solo ícono — iconOnly: true da padding cuadrado
new MTS.Button('#btn', {
  iconLeft: MTS.Icon.get('trash'),
  iconOnly: true,
  variant:  'danger',
})
4 — Estados
Disabled
Loading — se activa/desactiva con la API
Round — border-radius pill
Block — ancho completo
// Disabled
new MTS.Button('#btn', { label: 'Deshabilitado', disabled: true })
btn.disable()   // via API
btn.enable()    // via API

// Loading
const btn = new MTS.Button('#btn', { label: 'Guardar', variant: 'primary' })
btn.setLoading(true)                        // activa spinner
setTimeout(() => btn.setLoading(false), 2000)   // desactiva

// Round
new MTS.Button('#btn', { label: 'Pill button', round: true })

// Block
new MTS.Button('#btn', { label: 'Botón ancho completo', block: true })
5 — Grupos de botones
Selector de vista — activo al hacer click
Acciones — variantes mixtas
Tamaño sm — group con size
Paginación simple
// Selector con estado activo automático
const grupo = new MTS.ButtonGroup('#contenedor', [
  { label: 'Día',    variant: 'secondary' },
  { label: 'Semana', variant: 'secondary' },
  { label: 'Mes',    variant: 'primary'   },  // inicia activo
], { activeOnClick: true })

// Marcar activo programáticamente
grupo.setActive(2)          // activa el tercer botón
grupo.getButton(0).disable()  // deshabilita el primero

// Acciones con variantes mixtas
new MTS.ButtonGroup('#contenedor', [
  { label: 'Exportar', variant: 'ghost'  },
  { label: 'Duplicar', variant: 'ghost'  },
  { label: 'Eliminar', variant: 'danger' },
])

// Con tamaño uniforme
new MTS.ButtonGroup('#contenedor', [...botones], { size: 'sm' })
6 — Botones con menú contextual
Menú de acciones
Split button — acción principal + opciones
// Patrón: botón que abre/cierra un menú inline
function toggleMenu(menuId) {
  document.getElementById(menuId).classList.toggle('open')
}
// Cerrar al click fuera
document.addEventListener('click', (e) => {
  if (!e.target.closest('.btn-menu-wrap'))
    document.querySelectorAll('.btn-menu-list').forEach(m => m.classList.remove('open'))
})

// HTML del menú
<div class="btn-menu-wrap">
  <button class="mts-btn mts-btn--secondary" onclick="toggleMenu('mi-menu')">
    Acciones ▾
  </button>
  <div class="btn-menu-list" id="mi-menu">
    <button class="btn-menu-item">Editar</button>
    <button class="btn-menu-item">Duplicar</button>
    <div class="btn-menu-sep"></div>
    <button class="btn-menu-item danger">Eliminar</button>
  </div>
</div>
7 — Eventos onClick

Haz click en cualquier botón — el resultado aparece debajo en tiempo real. 

// onClick recibe el evento DOM y la instancia del botón
const btn = new MTS.Button('#btn', {
  label:   'Guardar cambios',
  variant: 'primary',
  onClick: (event, instance) => {
    instance.setLoading(true)
    instance.setLabel('Guardando...')

    setTimeout(() => {
      instance.setLoading(false)
      instance.setLabel('✓ Guardado')
      instance.setVariant('secondary')
    }, 1500)
  },
})

// También vía .on() — permite múltiples listeners
btn.on('click', (event, instance) => {
  console.log('click en:', instance.label)
})

// Evento DOM nativo
document.getElementById('btn')
  .addEventListener('mts:button:click', (e) => {
    console.log('instancia:', e.detail.button)
  })

9 — Shadow & Ring
// Solo sombra de color
new MTS.Button('#btn', { label: 'Shadow', shadow: true })

// Solo ring semitransparente
new MTS.Button('#btn', { label: 'Ring', ring: true })

// Ambos combinados
new MTS.Button('#btn', { label: 'Shadow + Ring', shadow: true, ring: true })

// Funciona en todas las variantes
new MTS.Button('#btn', { label: 'Danger', variant: 'danger', shadow: true, ring: true })
new MTS.Button('#btn', { label: 'Success', variant: 'success', shadow: true, ring: true })

// API dinámica
btn.setShadow(true)
btn.setRing(true)
