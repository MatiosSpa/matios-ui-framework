MTS.PhoneInput

Input de teléfono con selector de país, bandera y formato automático.

1 — Básico — Chile default
— interactúa —
<div id="phone-cl" data-label="Phone number" data-country="CL"> </div> <script> new MTS.PhoneInput('#phone-cl', { onChange: (e) => console.log(e.detail.full), onCountryChange: (e) => console.log('country:', e.detail.country), }); </script>
new MTS.PhoneInput('#el', { label: 'Teléfono', country: 'CL', onChange: ({ raw, formatted, full, country }) => { console.log(full) // +56 9 1234 5678 }, })
2 — Diferentes países iniciales
<div id="phone-mx" data-country="MX" data-label="México"></div> <div id="phone-ar" data-country="AR" data-label="Argentina"></div> <div id="phone-us" data-country="US" data-label="United States"></div> <script> new MTS.PhoneInput('#phone-mx'); new MTS.PhoneInput('#phone-ar'); new MTS.PhoneInput('#phone-us'); </script>
new MTS.PhoneInput('#el', { country:'MX', label:'México' }) new MTS.PhoneInput('#el', { country:'AR', label:'Argentina' }) new MTS.PhoneInput('#el', { country:'US', label:'EE.UU.' })
3 — Tamaños + API
— interactúa —
<div id="phone-api" data-label="Controlled phone" data-country="CL"> </div> <script> const ph = new MTS.PhoneInput('#phone-api', { onChange: (e) => console.log(e.detail.full), }); ph.getValue() // → { raw, formatted, full, country } ph.setValue('912345678') ph.setCountry('MX') ph.setError('Invalid number') ph.clearError() </script>
const ph = new MTS.PhoneInput('#el', { country:'CL' }) ph.getValue() // → { raw, formatted, full, country } ph.setCountry('MX') ph.setError('Número inválido') ph.clearError()

— ingresa número —
<div id="miPhone" data-label="Teléfono" data-country="CL" data-placeholder="9 1234 5678"></div> <script>new MTS.PhoneInput('#miPhone', { onChange: ({ full }) => console.log(full) })</script>
