MTS.FileUpload

Drag & drop file zone with image preview and validation.

1 — Imágenes con preview / Images with preview
— sube archivos para ver los callbacks —
<div id="upload-photos"></div> <script> new MTS.FileUpload('#upload-photos', { accept: 'image/*', multiple: true, maxSize: 2, maxFiles: 5, hint: 'PNG, JPG up to 2MB · Max 5 files', // Fires when file list changes / Se dispara al cambiar la lista onChange: (files) => console.log('files:', files.length), // Fires on add / Se dispara al agregar onAdd: (file) => console.log('added:', file.name), // Fires on remove / Se dispara al eliminar onRemove: (file) => console.log('removed:', file.name), // Fires on validation error / Se dispara en error de validación onError: (errs) => console.error(errs), }); </script>
new MTS.FileUpload('#upload-photos', { // Accepted types / Tipos aceptados accept: 'image/*', // Multiple files / Múltiples archivos multiple: true, // Max size per file in MB / Tamaño máximo por archivo en MB maxSize: 2, // Max number of files / Número máximo de archivos maxFiles: 5, // Show image thumbnails / Mostrar miniaturas de imágenes preview: true, hint: 'PNG, JPG up to 2MB · Max 5 files', // Fires when file list changes / Se dispara al cambiar la lista onChange: (files) => console.log('total:', files.length), // Fires when a file is added / Se dispara al agregar un archivo onAdd: (file) => console.log('added:', file.name), // Fires when a file is removed / Se dispara al eliminar un archivo onRemove: (file) => console.log('removed:', file.name), // Fires on validation errors / Se dispara en errores de validación onError: (errs) => console.error(errs), });
2 — Documento único / Single document
<div id="upload-cv"></div> <script> new MTS.FileUpload('#upload-cv', { accept: '.pdf', multiple: false, maxSize: 5, label: 'Upload your CV in PDF', hint: 'PDF up to 5MB', preview: false, onChange: (files) => console.log(files[0]?.name), }); </script>
new MTS.FileUpload('#upload-cv', { accept: '.pdf', multiple: false, // single file / archivo único maxSize: 5, // 5 MB preview: false, // no thumbnail / sin miniatura label: 'Upload your CV in PDF', hint: 'PDF up to 5MB', onChange: (files) => console.log(files[0]?.name), });
3 — Múltiples tipos / Multiple types + API
— usa los botones —
<div id="upload-docs"></div> <script> const fu = new MTS.FileUpload('#upload-docs', { accept: '.pdf,.doc,.docx,.xls,.xlsx', multiple: true, maxSize: 10, preview: false, hint: 'PDF, Word or Excel up to 10MB', }); fu.getFiles() // → File[] fu.clear() // clear all files / limpiar todos fu.open() // open selector / abrir selector </script>
const fu = new MTS.FileUpload('#upload-docs', { accept: '.pdf,.doc,.docx,.xls,.xlsx', multiple: true, maxSize: 10, preview: false, hint: 'PDF, Word or Excel up to 10MB', onChange: (files) => console.log('total:', files.length), }); // API fu.getFiles() // → File[] fu.clear() // clear all / limpiar todos fu.open() // open file selector / abrir selector
