MTS.Select

Select personalizado — básico, grupos, multi-select, autocomplete y cascada.

1 — Variantes básicas
new MTS.Select('#el', {
  label:'País', placeholder:'Seleccionar...',
  options:[{value:'cl',label:'Chile'},{value:'pe',label:'Perú'}],
  onSelect: (e) => console.log(e.detail.value, e.detail.text),
})
// API: sel.value / sel.text / sel.getValue() / sel.getText()
2 — Multi-select
new MTS.Select('#el', {
  multiple:true, maxSelect:3, value:['dev','ux'],
  options:[{value:'ux',label:'UX'},{value:'dev',label:'Dev'},{value:'qa',label:'QA'}],
  onSelect: (e) => console.log(e.detail.value, e.detail.text),
})
3 — Autocomplete

💡 Escribe: Apple, Samsung, Google, iPad

// Array local
new MTS.Select('#el', { searchable:true, minChars:0,
  onSearch: (q) => DATA.filter(o => o.name.includes(q)).map(o=>({value:o.id,label:o.name})) })
// Async con delay
new MTS.Select('#el', { searchable:true, minChars:1, debounce:300,
  onSearch: async (q) => { await delay(250); return filtrar(q); } })
4 — Cascada País → Estado → Ciudad
const selPais = new MTS.Select('#pais',{options:PAISES})
const selEstado = new MTS.Select('#estado',{options:[],disabled:true})
selPais.on('change', e => {
  selEstado.setOptions(ESTADOS[e.detail.value], true) // true=habilitar
})
