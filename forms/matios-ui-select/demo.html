MTS.Select

Select personalizado — básico, grupos, multi-select, autocomplete y cascada.

1 — Variantes básicas
<!-- Basic select / Select básico --> <div id="sel-country" data-label="Country" data-placeholder="Select a country..." data-clearable> </div> <script> new MTS.Select('#sel-country', { options: [ { value: 'cl', label: 'Chile' }, { value: 'ar', label: 'Argentina' }, { value: 'mx', label: 'México' }, ], onChange: (e) => console.log(e.detail.value, e.detail.text), }); </script>
// Basic / Básico const sel = new MTS.Select('#sel-country', { label: 'Country', placeholder: 'Select a country...', clearable: true, options: [ { value: 'cl', label: 'Chile' }, { value: 'ar', label: 'Argentina' }, { value: 'mx', label: 'México' }, ], // Fires on selection change / Se dispara al cambiar la selección onChange: (e) => console.log(e.detail.value, e.detail.text), }); // API sel.getValue() // → 'cl' sel.getText() // → 'Chile' sel.setValue('ar') sel.clear()
2 — Multi-select
<div id="sel-tags" data-label="Technologies" data-multiple data-searchable> </div> <script> new MTS.Select('#sel-tags', { options: [ { value: 'js', label: 'JavaScript' }, { value: 'ts', label: 'TypeScript' }, { value: 'css', label: 'CSS' }, ], value: ['js'], maxSelect: 3, onChange: (e) => console.log(e.detail.value), }); </script>
const sel = new MTS.Select('#sel-tags', { label: 'Technologies', multiple: true, maxSelect: 3, value: ['ux', 'dev'], options: [ { value: 'ux', label: 'UX Design' }, { value: 'dev', label: 'Development' }, { value: 'qa', label: 'QA Testing' }, { value: 'pm', label: 'Project Management' }, ], // Returns array / Retorna arreglo onChange: (e) => { console.log(e.detail.value); // → ['ux', 'dev'] console.log(e.detail.text); // → ['UX Design', 'Development'] }, });
3 — Autocomplete

💡 Escribe: Apple, Samsung, Google, iPad

<!-- Async search requires JavaScript / La búsqueda asíncrona requiere JavaScript --> <div id="sel-search"></div> <script> new MTS.Select('#sel-search', { label: 'Search products', searchable: true, minChars: 1, debounce: 300, // Called on each search / Se llama en cada búsqueda onSearch: async (query) => { const res = await fetch('/api/products?q=' + query); return res.json(); // [{ value, label }] }, onChange: (e) => console.log(e.detail.value), }); </script>
// Local search / Búsqueda local const DATA = [ { value: '1', label: 'iPhone 15' }, { value: '2', label: 'iPad Pro' }, { value: '3', label: 'Samsung Galaxy' }, { value: '4', label: 'Google Pixel' }, ]; new MTS.Select('#sel-autocomplete', { label: 'Search locally', searchable: true, minChars: 0, options: DATA, onChange: (e) => console.log(e.detail.value), }); // Async search / Búsqueda asíncrona new MTS.Select('#sel-async', { label: 'Search (async)', searchable: true, minChars: 1, debounce: 300, // Component calls this on each keystroke / El componente llama esto en cada tecla onSearch: async (query) => { await new Promise(r => setTimeout(r, 250)); // simulate API / simular API return DATA.filter(o => o.label.toLowerCase().includes(query.toLowerCase())); }, onChange: (e) => console.log(e.detail.value), });
4 — Cascada País → Estado → Ciudad
<!-- Three containers / Tres contenedores --> <div id="sel-country"></div> <div id="sel-state"></div> <div id="sel-city"></div> <script> const COUNTRIES = [ { value: 'cl', label: 'Chile' }, { value: 'ar', label: 'Argentina' }, ]; const STATES = { cl: [{ value: 'rm', label: 'Región Metropolitana' }, { value: 'bi', label: 'Biobío' }], ar: [{ value: 'ba', label: 'Buenos Aires' }, { value: 'cb', label: 'Córdoba' }], }; const selCountry = new MTS.Select('#sel-country', { label: 'Country', options: COUNTRIES }); const selState = new MTS.Select('#sel-state', { label: 'State', options: [], disabled: true }); const selCity = new MTS.Select('#sel-city', { label: 'City', options: [], disabled: true }); selCountry.onChange = null; // set via constructor below new MTS.Select('#sel-country', { label: 'Country', options: COUNTRIES, onChange: (e) => { selState.setOptions(STATES[e.detail.value] || [], true); // true = also enable selCity.setOptions([], false); selCity.disable(); }, }); </script>
const COUNTRIES = [ { value: 'cl', label: 'Chile' }, { value: 'ar', label: 'Argentina' }, ]; const STATES = { cl: [{ value: 'rm', label: 'Región Metropolitana' }, { value: 'bi', label: 'Biobío' }], ar: [{ value: 'ba', label: 'Buenos Aires' }, { value: 'cb', label: 'Córdoba' }], }; const selPais = new MTS.Select('#sel-pais', { label: 'País', options: COUNTRIES }); const selEstado = new MTS.Select('#sel-estado', { label: 'Estado', options: [], disabled: true }); const selCiudad = new MTS.Select('#sel-ciudad', { label: 'Ciudad', options: [], disabled: true }); selPais.on('change', (e) => { // setOptions(opts, enable) — second param enables the select // setOptions(opts, habilitar) — segundo param habilita el select selEstado.setOptions(STATES[e.detail.value] || [], true); selCiudad.setOptions([], false); selCiudad.disable(); }); selEstado.on('change', (e) => { selCiudad.setOptions([ { value: 'stg', label: 'Santiago' }, { value: 'pud', label: 'Pudahuel' }, ], true); });
